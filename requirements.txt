#
# Install with:
#     pip install -r requirements.txt
#
# Python 3.8 or newer is required: the core uses dataclasses and typing.Protocol.

# environment has been verified on the target machine.

pyvista>=0.44

acpype>=2020.0

#   OpenBabel (obabel; required to prepare downloaded PDB ligands as MOL2)
